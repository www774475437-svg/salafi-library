قاعده في الصبر لشيخ الاسلام ابن تيميه ابي العباس تقي الدين احمد بن عبد الحليم المتوفي سنه (٧٢٨ه) تحقيق اد. محمد بن خليفه التميمي الاستاذ في كليه الدعوه في الجامعه ال مقدمه ان الحمد لله نحمده ونستعينه ونستغفره، ونعوذ بالله من شرور انفسنا، ومن سييات اعمالنا، من يهده الله فلا مضل له، ومن يضلل فلا هادي له. واشهد ان لا اله الا الله وحده لا شريك له، واشهد ان محمدا عبده ورسوله. ﴿يا ايها الذين امنوا اتقوا الله حق تقاته ولا تموتن الا وانتم مسلمون﴾ [ال عمران ١٠٢] . ﴿يا ايها الناس اتقوا ربكم الذي خلقكم من نفس واحده وخلق منها زوجها وبث منهما رجالا كثيرا ونساء واتقوا الله الذي تساءلون به والارحام ان الله كان عليكم رقيبا﴾ [النساء ١] . ﴿يا ايها الذين امنوا اتقوا الله وقولوا قولا سديدا يصلح لكم اعمالكم ويغفر لكم ذنوبكم ومن يطع الله ورسوله فقد فاز فوزا عظيما﴾ [الاحزاب ٧٠-٧١] . اما بعد، فان اصدق الحديث كلام الله، وخير الهدي هدي محمد ﷺ، وشر الامور محدثاتها، وكل محدثه بدعه، وكل بدعه ضلاله. وبعد، فان الصبر من اعظم خصال الخير التي حث الله عليها في كتابه العظيم، وامر بها رسوله الكريم ﷺ في سنته المطهره، وقد وردت ماده (صبر) في القران الكريم في مايه واربعه مواضع، علي تنوع في مواردها واسباب ذكرها. فقد امر الله نبيه بخلق الصبر فقال: ﵁ ﴿واصبر وما صبرك الا بالله﴾ [النحل ١٢٧] وقال تعالي: ﴿فاصبر كما صبر اولوا العزم من الرسل﴾ [الاحقاف ٣٥] . وامر الله به المومنين، فقال تعالي: ﴿يا ايها الذين امنوا اصبروا وصابروا﴾
[ال عمران ٢٠٠] . واثني علي اهله، فقال تعالي: ﴿والصابرين في الباساء والضراء وحين الباس اوليك الذين صدقوا واوليك هم المتقون﴾ [البقره ١٧٧] . واخبر بمحبته للصابرين، فقال تعالي: ﴿والله يحب الصابرين﴾ [ال عمران ١٤٦]، ومعيته لهم، فقال تعالي: ﴿واصبروا ان الله مع الصابرين﴾ [الانفال ٤٦] . واخبر ان الصبر خير لاصحابه، فقال تعالي: ﴿ولين صبرتم لهو خير للصابرين﴾ [النحل ١٢٦] . ووعدهم ان يجزيهم اعلي واوفي واحسن مما عملوه، فقال تعالي: ﴿ولنجزين الذين صبروا اجرهم باحسن ما كانوا يعملون﴾ [النحل ٩٦] وقال تعالي: ﴿انما يوفي الصابرون اجرهم بغير حساب﴾ [الزمر ١٠] . وبشرهم فقال تعالي: ﴿وبشر الصابرين﴾ [البقره ١٥٥] واخبر ان جزاءهم الجنه فقال تعالي: ﴿وجزاهم بما صبروا جنه وحريرا﴾ [الانسان ١٢] . وقد قرن الله الصبر بالقيم العليا في الاسلام، فقرنه باليقين، قال تعالي: ﴿وجعلنا منهم ايمه يهدون بامرنا لما صبروا وكانوا باياتنا يوقنون﴾ [السجده ٢٤]، وقرنه بالتوكل، قال تعالي: ﴿نعم اجر العاملين الذين صبروا وعلي ربهم يتوكلون﴾ [العنكبوت ٥٨، ٥٩]، وقرنه بالصلاه في قوله تعالي: ﴿واستعينوا بالصبر والصلاه﴾ [البقره ١٥٣]، وقرنه بالتقوي في عده ايات منها: قوله تعالي: ﴿وان تصبروا وتتقوا فان ذلك من عزم الامور﴾ [ال عمران ١٨٦]، وفي قوله تعالي: ﴿وان تصبروا وتتقوا لا يضركم كيدهم شييا﴾ [ال عمران ١٢٠]، وقوله تعالي في سوره يوسف: ﴿انه من يتق ويصبر فان الله لا يضيع اجر المحسنين﴾ [يوسف ٩٠] . وقرن الله - ﵎ - الصبر بالعمل، فقال: ﴿الا الذين صبروا وعملوا الصالحات اوليك لهم مغفره واجر كبير﴾ [هود ١١]، وقرنه بالجهاد، في قوله
تعالي: ﴿ثم ان ربك للذين هاجروا من بعد ما فتنوا ثم جاهدوا وصبروا ان ربك من بعدها لغفور رحيم﴾ [النحل ١١٠]، وفي قوله تعالي: ﴿ولنبلونكم حتي نعلم المجاهدين منكم والصابرين﴾ [محمد ٣١]، وقرنه بالاستغفار: ﴿فاصبر ان وعد الله حق واستغفر لذنبك وسبح بحمد ربك بالعشي والابكار﴾، وقرنه بالتسبيح، في قوله تعالي: ﴿واصبر لحكم ربك فانك باعيننا وسبح بحمد ربك حين تقوم﴾ [الطور ٤٨] وفي قوله تعالي في سوره طه: ﴿فاصبر علي ما يقولون وسبح بحمد ربك قبل طلوع الشمس وقبل غروبها ومن اناء الليل فسبح واطراف النهار لعلك ترضي﴾ [طه ١٣٠]، وقرن الصبر في القران الكريم بالحق، في قوله تعالي: ﴿والعصر ان الاسان لفي خسر الا الذين امنوا وعملوا الصالحات وتواصوا بالحق وتواصوا بالصبر﴾ [العصر ١-٣]، وقرنه بالرحمه، قال تعالي: ﴿ثم كان من الذين امنوا وتواصوا بالصبر وتواصوا بالمرحمه﴾ [البلد ١٧]، وقرنه بالشكر في عده ايات، قال تعالي: ﴿ان في ذلك لايات لكل صبار شكور﴾ [ابراهيم ٥] . وحديث القران عن الصبر متنوع وممتع مما يدل علي اهميته ومكانته العظيمه، وكذا الشان في السنه النبويه، فقد حث النبي ﷺ امته علي هذا الخلق الكريم، وكانت سيرته ﷺ انموذجا يحتذي في التخلق بخلق الصبر بشتي انواعه واعلي درجاته، ومن قرا في سيرته العمليه وسنته القوليه سيجد ان للصبر شانا عظيما. ولقد وقفت علي مولف لطيف لشيخ الاسلام ابن تيميه -رحمه الله تعالي - في موضوع الصبر بعنوان (قاعده في الصبر) حوي - علي لطافته وصغر حجمه - فوايد غزيره وتاصيلات مفيده وبخاصه في جانب الصبر علي اذي الغير، صاغها المولف - ﵀ - باسلوب مميز فريد، جلي فيه مميزات هذا النوع من الصبر وبين فوايده، وما ان انتهيت من مطالعته وقراءته، حتي عقدت العزم علي تحقيقه واخراجه بغيه الانتفاع به في خاصه نفسي، وتسهيل
انتفاع عامه المسلمين بما حواه من فوايد وفرايد، وقد قدمت لهذا المولف بمقدمه في التعريف بهذه الرساله، ووصف نسخها الخطيه، والعمل الذي قمت به، وذلك علي النحو التالي: القسم الاول: في التعريف بالرساله ووصف نسخه الخطيه وعملي فيها. ا- التعريف بالرساله. اولا: اسم الرساله. ثانيا: توثيق نسبه الرساله الي المولف. ثالثا: موضوع الرساله. ب- وصف النسخ الخطيه. ج- عملي في الرساله. القسم الثاني: تحقيق النص. واذ اقدم هذه الرساله لارجو الله ان ينفع بها من اطلع عليها، وان يجعل عملي عملا صالحا لوجهه خالصا، والله الموفق والهادي الي سواء السبيل.
القسم الاول التعريف بالرساله ووصف النسخ الخطيه وعملي فيها اولا: التعريف بالرساله. ا- اسم الرساله: اسم الرساله كما هو مسجل في الصفحه الاولي من النسختين الخطيتين هو: (قاعده في الصبر) . ولكن جاء في رساله (اسماء مولفات شيخ الاسلام ابن تيميه) تاليف: محمد بن عبد الله بن احمد المعروف بابن رشيق المغربي (ت ٧٤) وكذا في العقود الدريه لابن عبد الهادي تسميتها ب: (قاعده في الصبر والشكر. نحو ستين ورقه) ١. وهذا الكلام يدل علي ان الرساله التي بين ايدينا هي جزء من مولف اكبر يشمل موضوعي الصبر والشكر، ويوكد ذلك مقدمه الرساله ونهايتها حيث ذكر المصنف الشكر فقال في المقدمه: "واذا اعتبر العبد الدين كله راه يرجع بجملته الي الصبر والشكر" وقال في نهايه الرساله: "والاصل الثاني الشكر وهو العمل بطاعه الله تعالي". ولكن لم اقف علي تكمله هذه الرساله. ب- توثيق نسبه الرساله الي المولف: يدل علي صحه نسبه الرساله الي شيخ الاسلام ابن تيميه امور منها: ١- ما جاء علي طره الرساله من قول الناسخ: (قاعده في الصبر للشيخ تقي الدين ابن تيميه ﵀) . ٢- ما جاء في رساله (اسماء مولفات شيخ الاسلام ابن تيميه) تاليف: _________ ١ انظر: الجامع لسيره شيخ الاسلام ابن تيميه (ص ٢٣) ط دار عالم الفوايد. وانظر: العقود الدريه (ص ٣٩) .
محمد بن عبد الله بن احمد المعروف بابن رشيق المغربي (ت ٧٤٩) حيث قال: (قاعده في الصبر والشكر. نحو ستين ورقه) ١. ٣ - ما جاء في كتاب العقود الدريه لابن عبد الهادي (ص ٣٩) حيث ذكر ان من مولفات ابن تيميه (قاعده في الصبر والشكر) . ج - موضوع الرساله: بدا المولف - ﵀ - هذه الرساله ببيان ان الدين كله يرجع بجملته الي امرين هما: الصبر والشكر، واستدل لذلك بقوله تعالي: ﴿ان في ذلك لايات لكل صبار شكور﴾ وبقوله ﷺ: "عجبا لامر المومن ان امره كله عجب، لا يقضي الله لمومن قضاء الا كان خيرا له، ان اصابته سراء شكر فكان خيرا له، وان اصابته ضراء صبر فكان خيرا له". ثم بين ان الصبر عموما ينقسم الي ثلاثه اقسام هي: اولا: صبر علي الطاعه حتي يفعلها. ثانيا: صبر عن المنهي عنه حتي لا يفعله. ثالثا: الصبر علي ما يصيبه بغير اختياره من المصايب. ثم بين ان المصايب نوعان: النوع الاول: نوع لا اختيار للخلق فيه، كالامراض وغيرها من المصايب السماويه، وهذا النوع يسهل الصبر فيه لان العبد يشهد فيه قضاء الله وقدره، وانه لا مدخل للناس فيه فيصبر اما اضطرارا واما اختيارا. والنوع الثاني: المصايب التي تحصل للعبد بفعل الناس، في ماله او عرضه او نفسه، وهذا النوع يصعب الصبر عليه جدا لان النفس تستشعر الموذي لها _________ ١ الرساله مطبوعه ضمن (الجامع لسيره شيخ الاسلام ابن تيميه) (ص ٢٣٦) ط دار عالم الفوايد.
وهي تكره الغلبه فتطلب الانتقام، ولا يصبر علي هذا النوع الا النبيون والصديقون. وقد اقتصر كلام المصنف - ﵀ - في بقيه الرساله علي الاسباب التي تعين العبد علي الصبر علي المصايب التي تصيبه بفعل الناس، وذكر ذلك من عشرين وجها. وختم المصنف كلامه بالاشاره الي الاصل الثاني وهو: الشكر وفسره بانه العمل بطاعه الله واقتصر علي ذلك وخلت الرساله من تفصيل القول في ذلك، ولعل السبب في ذلك هو تصرف من افرد الرساله بالذكر وفصلها عن باقي التصنيف والا فالرساله لها تتمه، ويشهد لذلك ما ذكره ابن رشيق في تعداده لمولفات ابن تيميه حيث قال: "قاعده في الصبر والشكر. نحو ستين ورقه" فقد تصرف المختصر في العنوان واقتصر كذلك علي ما كتب في موضوع الصبر فقط، ولم يكمل بقيه الرساله، والله اعلم. ولعل في ذلك ما يبرر كون الرساله قد خلت من الخاتمه وكتب بدلا من ذلك في اخرها "تمت بحمد الله وعونه". ب- وصف النسخ الخطيه: اعتمدت في تحقيق هذه الرساله علي نسختين خطيتين. النسخه الاولي: نسخه جامعه اكسفورد. وتقع ضمن مجموع فيه كتاب العبوديه لشيخ الاسلام ابن تيميه، ولها صوره فلميه في قسم المخطوطات بعماده شوون المكتبات بالجامعه الاسلاميه بالمدينه المنوره، برقم (ف ٨٥٧٠/٥) . وعدد لوحاتها: ٤ اربع لوحات. وعدد الاسطر: يبلغ ما بين عشرين وواحد وعشرين سطرا.
وعدد الكلمات: متوسط عدد الكلمات في كل سطر خمس عشره كلمه. اسم الناسخ: جاء في صفحه العنوان العباره التاليه: "هذه الرساله بخط العلامه بيدكين التركماني، تلميذ ابن تيميه مولف هذه الرساله". وقد ذكر الدكتور عبد الرحمن بن عبد الجبار الفريوايي في كتابه "شيخ الاسلام ابن تيميه وجهوده في الحديث وعلومه": "ان من تلاميذ ابن تيميه ادريس ابن بيدكين بن عبد الله التركي الحنفي مولف كتاب اللمع في الحوادث والبدع. تتلمذ علي شيخ الاسلام"١. تاريخ النسخ: جاء في اخر النسخه: "تمت - بحمد الله تعالي وعونه - في يوم الاثنين سابع جمادي الاخره سنه ثمان وثمانمايه". نوع الخط ووصفه: كتبت هذه النسخه بخط نسخ عادي منقوط، واضح ومقروء، وهي سالمه من الخرم والعيوب، ولكنها خاليه من المقابله او السماعات. وقد رمزت لهذه النسخه بالرمز (ا) . النسخه الثانيه: نسخه جامعه برنستون بنيوجرسي بالولايات المتحده الامريكيه. وهي في ضمن مجموع فيه كتاب (العرش) للذهبي، و(لمعه الاعتقاد) لابن قدامه و(فصل في شرح حديث سيد الاستغفار) لشيخ الاسلام ابن تيميه، ولها صوره فيلميه في قسم المخطوطات بعماده شوون المكتبات بالجامعه الاسلاميه بالمدينه المنوره برقم (٩٧٤٢/٣) . وعدد لوحاتها: ٨ ثماني لوحات. وعدد الاسطر: ١٣ ثلاثه عشر سطرا. _________ (١/١١٨)
وعدد الكلمات: متوسط عدد الكلمات في السطر اثنتا عشره كلمه. اسم الناسخ: محمد بن اسحاق التميمي داري. تاريخ النسخ: لم يذكر. نوع الخط ووصفه: كتبت هذه النسخه بخط نسخ عادي منقوط واضح ومقروء وسالم من الخرم والعيوب، وخاليه من المقابله والسماعات. وقد رمزت لهذه النسخه بالرمز (ب) . وهناك نسخه ثالثه لهذه الرساله وقفت علي معلومات عنها ضمن (ثبت فيه قوايم ببعض مخطوطات شيخ الاسلام ابن تيميه) تصنيف الشيخ علي بن عبد العزيز الشبل. جاء فيه (ص ٩١): "رساله في الصبر في ٨ ورقات بمكتبه الاوقاف العامه ببغداد، برقم (٤٧١٥/٢)، كتبها سلطان بن ملا حسين القره غول، سنه ١٢١٧؟". ولم استطع الوقوف عليها للاوضاع الراهنه في العراق. ج- عملي في الكتاب: ١- نسخت النص وفق القواعد الاملاييه الحديثه، مع الضبط بالشكل لما قد يشكل. ٢- عزوت الايات القرانيه الي سورها مع ذكر رقمها في الحاشيه. ٣- قمت بتخريج الاحاديث النبويه المرفوعه مع بيان حكم اهل العلم عليها، كما اجتهدت في تخريج الاثار الموقوفه حسب ما وقفت عليه. ٤- شرحت ما يحتاج لشرح من الالفاظ الغريبه الوارده في النص. ٥- وضعت للكتاب مقدمه فيها تعريف بالكتاب ومحتواه، وبالنسخ الخطيه، وعملي فيه.

بسم الله الرحمن الرحيم اللهم اعف واغفر. قال الشيخ الامام العامل شيخ الاسلام، مفتي الانام تقي الدين ابو العباس احمد ابن تيميه١. فصل: جعل الله - ﷾ - عباده المومنين بكل منزله خيرا منه، فهم دايما في نعمه من ربهم، اصابهم ما يحبون، او ما يكرهون، وجعل اقضيته واقداره التي يقضيها لهم ويقدرها عليهم متاجر يربحون بها عليه، وطرقا يصلون منها اليه، كما ثبت في الصحيح عن امامهم ومتبوعهم الذين اذا دعي يوم القيامه كل اناس بامامهم دعوا به - صلوات الله وسلامه عليه -٢ _________ ١ افتتحت نسخه (ب) بعد البسمله بالعباره التاليه: "وصلي الله علي سيدنا محمد، قال الشيخ الامام العالم العلامه شيخ الاسلام، مفتي الانام تقي الدين ابو العباس احمد بن تيميه الحراني ﵁". ٢ اخرج الخطيب في تاريخ بغداد (١/٣١٧) عن انس في قوله تعالي: ﴿يوم ندعو كل اناس بامامهم﴾ الايه [٧١] من سوره الاسراء قال: "نبيهم"، واورده السيوطي في الدر المنثور (٥/٣١٦) وعزاه لابن ابي حاتم وابن مردويه. وروي مثله عن مجاهد اخرجه ابن جرير في تفسيره (١٥/١٢٦)، واورده السيوطي في الدر المنثور (٥/٣١٦) وعزاه لابن المنذر.
انه قال: "عجبا لامر المومن ان امره كله عجب لا١يقضي الله لمومن قضاء الا كان خيرا له ان اصابته سراء شكر فكان خيرا له وان اصابته ضراء صبر فكان خيرا له"٢. فهذا الحديث يعم جميع اقضيته لعبده المومن وانها خيرله اذا صبر علي مكروهها وشكر لمحبوبها، بل هذا داخل في مسمي الايمان كما٣ قال بعض السلف: "الايمان نصفان نصف صبر ونصف شكر"٤ لقوله تعالي: ﴿ان في ذلك _________ ١ في (ب): (ما) . ٢ اخرجه مسلم،كتاب الزهد، باب المومن امره كله خير (٨/٢٢٧)، واخرجه الامام احمدفي المسند (١/١٧٣)، واخرجه الدارمي في سننه، كتاب الرقايق،باب المومن يوجر في كل شيء (٢/٣١٨)، بلفظ: "عجبا لامر المومن ان امره كله خير ... الحديث". ولم اقف علي الحديث بلفظ: " ... كله عجب" كما اورده المولف. ٣ في (ب): "فانه كما قال ... " ٤ اخرجه وكيع بن الجراح في الزهد (٢/٤٥٦) برقم (٢٠٣)،والطبراني في الكبير (٩/١٠٧)، والمنذري في الترغيب والترهيب (٤/١٧١ ح ٤٩٧٦)، موقوفا علي ابن مسعود بلفظ: "الصبر نصف الايمان، واليقين الايمان كله". والبخاري في صحيحه (١/٤٥) كتاب الايمان باب قول النبي ﷺ: "بني الاسلام علي خمس" تعليقا واقتصر علي شطر منه من قول ابن مسعود بلفظ: "اليقين الايمان كله". والخرايطي في فضيله الشكر لله علي نعمته (ص ٣٩) برقم (١٨) بلفظ: "الايمان نصفان: فنصف في الصبر ونصف في الشكر". وابن الاعرابي في معجمه (١/٣٠٩)، وابو نعيم في الحليه (٥/٣٤)، وتمام في فوايده (٢/٤٠)، وابو الحسن الازدي في المجلس الاول من المجالس الخمسه (١٦-١٧)، والقضاعي في مسند الشهاب (١/١٢٦ - ١٢٧)، والبيهقي في شعب الايمان (٧/١٢٣) برقم (٩٧١٥)،والاصبهاني في الترغيب والترهيب (٢/٦٦١) برقم (١٥٨٢)، والخطيب في تاريخ بغداد (١٣/٢٢٦)، وابن الجوزي في العلل المتناهيه (٢/٣٣٠-٣٣١)، والذهبي في مختصر العلل (٣/١١٢٥)، مرفوعا بلفظ: "الصبر نصف الايمان، واليقين الايمان كله" وابن ابي الدنيا في الشكر برقم (١٤) بلفظ: "الشكر نصف الايمان، والصبر نصف الايمان، واليقين الايمان كله". والبيهقي في الزهد (١/٢٨) موقوفا علي ابن مسعود: "الايمان نصفان، نصف في الصبر ونصف في الشكر".وقال: "قد روي هذا من وجه اخر غير قوي مرفوعا"، وقال في الشعب: "والمحفوظ عن ابن مسعود من قوله غير مرفوع". واورده الهيثمي في مجمع الزوايد (١/٥٧) وقال عن اسناد الطبراني: "رجاله رجال الصحيح". والسيوطي في الجامع الصغير (مع الفيض ٤/٢٣٣) ورمز لضعفه، واعله المناوي بيعقوب بن حميد، وهو ضعيف من قبيل حفظه. وقال الحافظ في الفتح (١/٤٨): "هذا التعليق طرف من اثر وصله الطبراني بسند صحيح، وبقيته: والصبر نصف الايمان". وقال: "واخرجه ابو نعيم والبيهقي من حديثه (اي ابن مسعود) في الزهد مرفوعا ولا يثبت رفعه". وقال في اللسان (٥/١٥٢): "قال ابو علي النيسابوري: هذا الحديث منكر، لا اصل له من حديث زبيد ولا من حديث الثوري". ولم يحسنه غير العراقي في تخريج الاحياء (١/٧٢، ١/٢٣١) . وقال عنه الالباني كما في سلسله الاحاديث الضعيفه (١/٥٠٦-٥٠٧) برقم (٤٩٩):"منكر". وخلاصه القول في هذا الاثر انه صحيح موقوفا، وضعيف مرفوعا، وقول ابن تيميه هنا "قال بعض السلف" يويد بكونه موقوفا.
لايات لكل صبار شكور﴾ ١ واذا اعتبر العبد الدين كله راه يرجع بجملته الي الصبر والشكر، وذلك لان الصبر ثلاثه اقسام: صبر علي الطاعه حتي يفعلها، فان العبد لا يكاد يفعل المامور به الا بعد _________ ١ الايه [٥] من سوره ابراهيم والايه [٣١] من سوره لقمان والايه [١٩] من سوره سبا والايه [٣٣] من سوره الشوري
صبر ومصابره ومجاهده لعدوه الباطن والظاهر١، فبحسب هذا الصبر يكون اداوه للمامورات وفعله للمستحبات. النوع الثاني: صبر عن المنهي عنه٢ حتي لا يفعله، فان النفس ودواعيها، وتزيين الشيطان، وقرناء السوء، تامره بالمعصيه وتجريه عليها، فبحسب قوه صبره يكون تركه لها، قال بعض السلف: اعمال البر يفعلها البر والفاجر ولا يقدر علي ترك المعاصي الا صديق٣. النوع الثالث: الصبر علي ما يصيبه بغير اختياره من المصايب وهي نوعان: نوع لا اختيار للخلق فيه، كالامراض وغيرها من المصايب السماويه، فهذه يسهل الصبر فيها، لان العبد يشهد فيها قضاء الله وقدره، وانه لا مدخل للناس فيها، فيصبر اما اضطرارا، واما اختيارا، فان فتح الله علي قلبه باب الفكره في فوايدها وما في حشوها٤ من النعم والالطاف٥ انتقل من الصبر عليها الي الشكر _________ ١ في (ب): "الظاهر والباطن ... ". (عنه) ساقطه من (ب) . ٣ اخرجه ابو نعيم في حليه الاولياء (١٠ /١٩٧) باسناده عن سهل بن عبد الله التستري ضمن كلام له طويل بلفظ "ليس من عمل بطاعه الله صار حبيب الله، ولكن من اجتنب ما نهي الله عنه صار حبيب الله، ولا يجتنب الاثام الا صديق مقرب، واما اعمال البر يعملها البر والفاجر". ٤ معني الحشي: الناحيه، والمقصود نواحيها او داخلها. انظر: لسان العرب ماده (حشا) (١٤/١٨٠-١٨١) . ٥ اللطيف من الكلام: ما غمض معناه وخفي، واللطف في العمل: الرفق فيه. انظر: لسان العرب ماده (لطف) (٩/٣١٦) .

فهذا النوع يصعب الصبر عليه جدا، لان النفس تستشعر الموذي لها١، وهي تكره الغلبه، فتطلب الانتقام، فلا يصبر علي هذا النوع الا الانبياء والصديقون، وكان نبينا ﷺ اذا اوذي يقول: يرحم الله موسي لقد اوذي باكثر من هذا فصبر٢ واخبر عن نبي من الانبياء انه ضربه قومه فجعل يقول: "اللهم اغفر لقومي فانهم لا يعلمون" ٣ وقد روي عنه ﷺ انه جري له هذا مع قومه [فجعل يقول مثل ذلك] ٤، فجمع في هذا ثلاثه امور: العفو عنهم، والاستغفار لهم، والاعتذار عنهم بانهم لا يعلمون، وهذا النوع من الصبر عاقبته النصر والعز٥ والسرور والامن والقوه في ذات الله، وزياده محبه الله ومحبه الناس له وزياده العلم، ولهذا قال الله تعالي: ﴿وجعلنا منهم ايمه يهدون بامرنا لما _________ (لها) ساقطه من (ب) . ٢ اخرجه البخاري في صحيحه، كتاب احاديث الانبياء، باب (٢٨)، رقم (٣٤٠٥) . ومسلم في صحيحه، كتاب الزكاه، باب اعطاء المولفه قلوبهم (٣/١٠٩) . ٣ اخرجه البخاري في صحيحه، كتاب احاديث الانبياء، باب (٥٤) رقم (٣٤٧٧) . ومسلم في صحيحه، كتاب الجهاد، باب غزوه احد (٥/١٧٩) . ٤ ساقط من (ا) . وهذا الحديث رواه عبد الله بن مسعود ﵁ قال: "لما قسم رسول الله ﷺ غنايم حنين بالجعرانه ازدحموا عليه فقال رسول الله ﷺ: "ان عبدا من عباد الله بعثه الله الي قومه فضربوه وشجوه قال: فجعل يمسح الدم عن جبهته ويقول: رب اغفر لقومي فانهم لا يعلمون. قال عبد الله: كاني انظر الي رسول الله ﷺ يمسح الدم عن جبهته يحكي الرجل ويقول: رب اغفر لقومي فانهم لا يعلمون"، اخرجه الامام احمد في المسند (٧/ ٣٧٦) برقم (٤٣٦٦)،ط. موسسه الرساله وقال محققه: صحيح لغيره. ٥ في (ب): (والنصر والهدي) .
صبروا وكانوا باياتنا يوقنون﴾ ١ فبالصبر واليقين تنال٢ الامامه في الدين، فاذا انضاف الي هذا الصبر قوه اليقين والايمان ترقي العبد في درجات السعاده بفضل الله، وذلك فضل الله يوتيه من يشاء والله ذو الفضل العظيم. ولهذا قال الله تعالي: ﴿ادفع بالتي هي احسن فاذا الذي بينك وبينه عداوه كانه ولي حميم وما يلقاها﴾ يعني: الاعمال الصالحه مثل العفو والصفح٣ ﴿الا الذين صبروا وما يلقاها الا ذو حظ عظيم﴾ ٤ نصيب وافر وهي الجنه٥. ويعين العبد علي هذا الصبر عده اشياء: _________ ١ الايه [٢٤] من سوره السجده. ٢ في (ب): (فالصبر واليقين ينال) . ٣ جمله "يعني الاعمال الصالحه مثل العفو والصفح" ساقطه من (ب) . ٤ الايتان [٣٤،٣٥] من سوره فصلت.

سلطهم عليه١، عن ذمهم ولومهم والوقيعه فيهم، واذا رايت العبد يقع في الناس اذا اذوه ولا يرجع الي نفسه باللوم والاستغفار فاعلم ان مصيبته مصيبه حقيقيه، واذا تاب واستغفر، وقال: هذا بذنوبي، صارت في حقه نعمه. قال علي بن ابي طالب - ﵁٢ كلمه من جواهر الكلام: لا يرجون عبد الا ربه، ولا يخافن عبد الا ذنبه ٣ وروي عنه وعن غيره: ما نزل بلاء الا بذنب، ولا رفع الا بتوبه٤٥. _________ ١ هكذا في الاصل ولعل الصواب (التي سلطها عليه) . ٢ في (ب): (كرم الله وجهه) . ٣ اخرجه البيهقي في شعب الايمان (٧/١٢٢) رقم (٩٧١٨) . والاصبهاني في الترغيب والترهيب (٢/٦٦٢) رقم (١٥٨٦) . وابن عبد البر في جامع بيان العلم وفضله (١/٣٨٣) رقم (٥٤٧-٥٤٨) . ٤ رواه الزبير بن بكار في الانساب (كما عزاه اليه الحافظ ابن حجر في الفتح ٢/٤٩٧ ولم اجده في القسم المطبوع من كتاب الانساب) ان العباس لما استسقي به عمر قال: "اللهم انه لم ينزل بلاء الا بذنب، ولم يكشف الا بتوبه" ضمن دعاء طويل. واخرجه ابوبكر احمد بن مروان الدينوري (ت ٣٣٣) في المجالسه وجواهر العلم (٣/١٠٢ - ١٠٣ ط. دار ابن حزم) موقوفا علي العباس في دعايه عندما استسقي به عمر، فكان من دعايه "اللهم انه لم ينزل بلاء من السماء الا بذنب، ولا يكشف الا بتوبه". كما اخرجه ابن عساكر في تاريخ دمشق (٢٦/ ٣٥٨ - ٣٥٩ ط دار الفكر، ص ١٨٤ تراجم عباده بن اوفي - عبد الله بن ثوب) . من طريق ابي صالح باذام مولي ابن هانيء عن العباس بن عبد المطلب في حديث طويل. وقد ثبت في صحيح البخاري استسقاء عمر بدعاء العباس (٢/٤٩٤ مع الفتح) كتاب الاستسقاء - باب سوال الناس الامام الاستسقاء اذا قحطوا - ح ١٠١٠ دون بيان صفه دعايه علي وجه التفصيل، فلم يرد ذكر هذا اللفظ. وانظر الكلام علي هذا الحديث في اقتضاء الصراط المستقيم (ص ٣٣٧، ٣٣٨، ط دار الافتاء) .
